   220→    try {
   221→      const payload = {
   222→        name: `${product.name} (Copy)`,
   223→        description: product.description,

   232→        images: product.images,

   274→      toast.error('Error deleting product');
   275→    }

   292→        method,
   293→        headers: { 'Content-Type': 'application/json' },
   294→        body: JSON.stringify(payload),
   295→      });

   297→      if (res.ok) {

   308→        toast.error(err.error || 'Failed to save product');
   309→      }
   310→    } catch {
   311→      toast.error('Error saving product');
   312→    } finally {
   313→      setIsSubmitting(false);
   314→    }

   336→      toast.error('No valid URLs found. Paste one URL per line.');
   337→    }
   338→  };

   341→    setForm((prev) => ({ ...prev, images: prev.images.filter((_, i) => i !== index) }));
   342→  };

   347→      ...prev,
   348→      variants: [...prev.variants, { name: '', color: '', size: '', material: '', price: '', stock: '10' }],

   350→  };

   360→

   363→    setForm((prev) => ({ ...prev, attributes: [...prev.attributes, { name: '', value: '' }] }));

   365→  const removeAttribute = (index: number) => {
   366→    setForm((prev) => ({ ...prev, attributes: prev.attributes.filter((_, i) => i !== index) }));
   367→  };
   368→  const updateAttribute = (index: number, field: keyof AttributeRow, value: string) => {
   369→    setForm((prev) => ({
   370→      ...prev,
   371→      attributes: prev.attributes.map((a, i) => (i === index ? { ...a, [field]: value } : a)),
   372→    }));
   373→  };
   374→
   375→  // ---- Tag toggle ----
   376→  const toggleTag = (tagId: string) => {

   378→      ...prev,
   379→      selectedTags: prev.selectedTags.includes(tagId)
   380→        ? prev.selectedTags.filter((t) => t !== tagId)
   381→        : [...prev.selectedTags, tagId],
   382→    }));
   383→  };
   384→
   385→  const selectedCategory = categories.find((c) => c.id === form.categoryId);
   386→  const subCategories = selectedCategory?.subCategories || [];
   387→
   388→  // ---- Brand form ----
   389→  const [brandForm, setBrandForm] = useState({ name: '', description: '', country: '', foundedYear: '' });
   390→  const [showBrandForm, setShowBrandForm] = useState(false);
   391→

   393→    if (!brandForm.name.trim()) { toast.error('Brand name is required'); return; }
   394→    try {
   395→      const res = await fetch('/api/admin/brands', {
   396→        method: 'POST',
   397→        headers: { 'Content-Type': 'application/json' },
   398→        body: JSON.stringify(brandForm),
   399→      });
   400→      if (res.ok) {
   401→        toast.success('Brand created successfully');
   402→        const newBrands = await (await fetch('/api/brands')).json();
   403→        setBrands(newBrands);
   404→        setBrandForm({ name: '', description: '', country: '', foundedYear: '' });

   408→  };

   418→        method: 'POST',
   419→        headers: { 'Content-Type': 'application/json' },
   420→        body: JSON.stringify(categoryForm),

   422→      if (res.ok) {

   443→                <ArrowLeft className="h-5 w-5" />
   444→              </button>
   445→              <div>
   446→                <h1 className="text-2xl font-light tracking-wide flex items-center gap-3">
   447→                  <Settings className="h-5 w-5 text-gold" />

   492→        {/* ===================== PRODUCTS TAB ===================== */}

   504→                >
