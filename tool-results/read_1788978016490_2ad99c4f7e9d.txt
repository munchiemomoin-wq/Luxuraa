    35→
    36→export function AdminPanel() {
    37→  const {
    38→    brands,
    39→    categories,
    40→    tags,
    41→    setBrands,
    42→    setCategories,
    43→    setProducts,
    44→    navigateTo,
    45→    logoutAdmin,
    46→  } = useStore();
    47→
    48→  // ---- Admin-specific product state (independent of store) ----
    49→  const [adminProducts, setAdminProducts] = useState<Product[]>([]);
    50→  const [adminTotal, setAdminTotal] = useState(0);
    51→  const [adminPage, setAdminPage] = useState(1);
    52→  const [isLoadingAdmin, setIsLoadingAdmin] = useState(false);
    53→  const [adminSearch, setAdminSearch] = useState('');
    54→
    55→  const fetchAdminProducts = useCallback(async (page: number, append: boolean = false, search?: string) => {
    56→    setIsLoadingAdmin(true);
    57→    try {
    58→      const params = new URLSearchParams({
    59→        page: String(page),
    60→        limit: String(ADMIN_PAGE_SIZE),
    61→        sortBy: 'createdAt',
    62→        sortOrder: 'desc',
    63→      });
    64→      if (search) params.set('search', search);
    65→
    66→      const res = await fetch(`/api/products?${params.toString()}`);
    67→      if (res.ok) {
    68→        const data = await res.json();
    69→        setAdminProducts((prev) => append ? [...prev, ...data.products] : data.products);
    70→        setAdminTotal(data.pagination.total);
    71→      }
    72→    } catch {
    73→      // silent
    74→    } finally {
    75→      setIsLoadingAdmin(false);
    76→    }
    77→  }, []);
    78→
    79→  // Initial load + when search changes
    80→  useEffect(() => {
    81→    setAdminPage(1);
    82→    fetchAdminProducts(1, false, adminSearch || undefined);
    83→  }, [adminSearch, fetchAdminProducts]);
    84→

    88→      const res = await fetch('/api/products?limit=12&sortBy=createdAt&sortOrder=desc');
    89→      if (res.ok) {
    90→        const data = await res.json();
    91→        setProducts(data.products, data.pagination);
    92→      }
    93→    } catch { /* silent */ }
    94→  };
    95→
    96→  const loadMore = () => {
    97→    const nextPage = adminPage + 1;
    98→    setAdminPage(nextPage);
    99→    fetchAdminProducts(nextPage, true, adminSearch || undefined);
   100→  };
   101→
   102→  // Filtered products shown in table
   103→  const displayedProducts = adminSearch
   104→    ? adminProducts
   105→    : adminProducts;
   106→
   107→  // ---- Tabs ----
   108→  const [activeTab, setActiveTab] = useState<'products' | 'brands' | 'categories'>('products');
   109→
   110→  // ---- Product form ----
   111→  const [showForm, setShowForm] = useState(false);
   112→  const [editingId, setEditingId] = useState<string | null>(null);
   113→  const [isSubmitting, setIsSubmitting] = useState(false);
   114→  const [isUploading, setIsUploading] = useState(false);
   115→  const formRef = useRef<HTMLDivElement>(null);
   116→  const fileInputRef = useRef<HTMLInputElement>(null);
   117→
   118→  const [form, setForm] = useState({
   119→    name: '',
   120→    description: '',
   121→    price: '',
   122→    compareAtPrice: '',
   123→    brandId: '',
   124→    categoryId: '',
   125→    subCategoryId: '',
   126→    featured: false,
   127→    bestseller: false,
   128→    newArrival: false,
   129→    images: [] as string[],
   130→    videoUrl: '',
   131→    variants: [{ name: 'Default', color: '', size: '', material: '', price: '', stock: '10' }] as VariantRow[],
   132→    attributes: [{ name: '', value: '' }] as AttributeRow[],
   133→    selectedTags: [] as string[],
   134→  });

   136→  const resetForm = () => {
   137→    setForm({
   138→      name: '',
   139→      description: '',
   140→      price: '',
   141→      compareAtPrice: '',
   142→      brandId: '',
   143→      categoryId: '',
   144→      subCategoryId: '',
   145→      featured: false,
   146→      bestseller: false,
   147→      newArrival: false,
   148→      images: [],
   149→      videoUrl: '',
   150→      variants: [{ name: 'Default', color: '', size: '', material: '', price: '', stock: '10' }],
   151→      attributes: [{ name: '', value: '' }],
   152→      selectedTags: [],
   153→    });
   154→    setEditingId(null);
   155→    setShowForm(false);
   156→  };
   157→
   158→  const keepFormForNext = () => {
   159→    const currentBrand = form.brandId;
   160→    const currentCategory = form.categoryId;
   161→    const currentSubCategory = form.subCategoryId;
   162→    const currentTags = form.selectedTags;
   163→    const currentAttributes = form.attributes;
   164→    setForm({
   165→      name: '',
   166→      description: '',
   167→      price: '',
   168→      compareAtPrice: '',
   169→      brandId: currentBrand,
   170→      categoryId: currentCategory,
   171→      subCategoryId: currentSubCategory,
   172→      featured: false,
   173→      bestseller: false,
   174→      newArrival: false,
   175→      images: [],
   176→      videoUrl: '',
   177→      variants: [{ name: 'Default', color: '', size: '', material: '', price: '', stock: '10' }],
   178→      attributes: currentAttributes.filter(a => a.name.trim()),
   179→      selectedTags: currentTags,
   180→    });
   181→    setEditingId(null);
   182→    // Keep form open for rapid entry
   183→  };
   184→
   185→  const handleEditProduct = (product: any) => {
   186→    setForm({
   187→      name: product.name,
   188→      description: product.description || '',
   189→      price: String(product.price),
   190→      compareAtPrice: product.compareAtPrice ? String(product.compareAtPrice) : '',
   191→      brandId: product.brandId,
   192→      categoryId: product.categoryId,
   193→      subCategoryId: product.subCategoryId || '',
   194→      featured: product.featured,
   195→      bestseller: product.bestseller,
   196→      newArrival: product.newArrival,
   197→      images: (() => { try { return JSON.parse(product.images); } catch { return []; } })(),
   198→      videoUrl: (product as any).videoUrl || '',
   199→      variants: product.variants.length > 0
   200→        ? product.variants.map((v: any) => ({
   201→            name: v.name,
   202→            color: v.color || '',
   203→            size: v.size || '',
   204→            material: v.material || '',
   205→            price: v.price ? String(v.price) : '',
   206→            stock: String(v.stock),
   207→          }))
   208→        : [{ name: 'Default', color: '', size: '', material: '', price: '', stock: '10' }],
   209→      attributes: product.attributes.length > 0
   210→        ? product.attributes.map((a: any) => ({ name: a.name, value: a.value }))
   211→        : [{ name: '', value: '' }],
   212→      selectedTags: product.tags.map((t: any) => t.tagId),
   213→    });
   214→    setEditingId(product.id);
   215→    setShowForm(true);
   216→    setTimeout(() => formRef.current?.scrollIntoView({ behavior: 'smooth' }), 100);
   217→  };
   218→
   219→  const handleDuplicateProduct = async (product: any) => {
   220→    try {
   221→      const payload = {
   222→        name: `${product.name} (Copy)`,
   223→        description: product.description,
   224→        price: String(product.price),
   225→        compareAtPrice: product.compareAtPrice ? String(product.compareAtPrice) : '',
   226→        brandId: product.brandId,
   227→        categoryId: product.categoryId,
   228→        subCategoryId: product.subCategoryId,
   229→        featured: false,
   230→        bestseller: false,
   231→        newArrival: false,
   232→        images: product.images,
   233→        variants: product.variants.map((v: any) => ({
   234→          name: v.name,
   235→          color: v.color || '',
   236→          size: v.size || '',
   237→          material: v.material || '',
   238→          price: v.price ? String(v.price) : '',
   239→          stock: String(v.stock),
   240→        })),
   241→        attributes: product.attributes.map((a: any) => ({ name: a.name, value: a.value })),
   242→        tags: product.tags.map((t: any) => t.tagId),
   243→      };
   244→
   245→      const res = await fetch('/api/admin/products', {
   246→        method: 'POST',
   247→        headers: { 'Content-Type': 'application/json' },
   248→        body: JSON.stringify(payload),
   249→      });
   250→      if (res.ok) {
   251→        toast.success(`Duplicated "${product.name}"`);
   252→        fetchAdminProducts(1, false, adminSearch || undefined);
   253→        refreshStoreProducts();
   254→      } else {
   255→        toast.error('Failed to duplicate');
   256→      }
   257→    } catch {
   258→      toast.error('Error duplicating product');
   259→    }
   260→  };
   261→
   262→  const handleDeleteProduct = async (id: string) => {
   263→    if (!confirm('Are you sure you want to delete this product?')) return;
   264→    try {
   265→      const res = await fetch(`/api/admin/products?id=${id}`, { method: 'DELETE' });
   266→      if (res.ok) {
   267→        toast.success('Product deleted successfully');
   268→        fetchAdminProducts(1, false, adminSearch || undefined);
   269→        refreshStoreProducts();
   270→      } else {
   271→        toast.error('Failed to delete product');
   272→      }
   273→    } catch {
   274→      toast.error('Error deleting product');
   275→    }
   276→  };
   277→
   278→  const handleSubmit = async (andAddAnother: boolean = false) => {
   279→    if (!form.name.trim()) { toast.error('Product name is required'); return; }
   280→    if (!form.price) { toast.error('Price is required'); return; }
   281→    if (!form.brandId) { toast.error('Please select a brand'); return; }
   282→    if (!form.categoryId) { toast.error('Please select a category'); return; }
   283→    if (form.images.length === 0) { toast.error('Please upload at least one image'); return; }
   284→
   285→    setIsSubmitting(true);
   286→    try {
   287→      const payload: any = { ...form, tags: form.selectedTags, videoUrl: form.videoUrl || null };
   288→      const method = editingId ? 'PUT' : 'POST';
   289→      if (editingId) payload.id = editingId;
   290→
   291→      const res = await fetch('/api/admin/products', {
   292→        method,
   293→        headers: { 'Content-Type': 'application/json' },
   294→        body: JSON.stringify(payload),
   295→      });
   296→
   297→      if (res.ok) {
   298→        toast.success(editingId ? 'Product updated!' : 'Product created!');
   299→        if (andAddAnother) {
   300→          keepFormForNext();
   301→        } else {
   302→          resetForm();
   303→        }
   304→        fetchAdminProducts(1, false, adminSearch || undefined);
   305→        refreshStoreProducts();
   306→      } else {
   307→        const err = await res.json();
   308→        toast.error(err.error || 'Failed to save product');
   309→      }
   310→    } catch {
   311→      toast.error('Error saving product');
   312→    } finally {
   313→      setIsSubmitting(false);
   314→    }
   315→  };
   316→
   317→  // ---- Image file upload ----
   318→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   319→    const files = e.target.files;
   320→    if (!files || files.length === 0) return;
   321→
   322→    const remaining = 6 - form.images.length;
   323→    if (remaining <= 0) {
   324→      toast.error('Maximum 6 images allowed');
   325→      return;
   326→    }
   327→
   328→    const filesToUpload = Array.from(files).slice(0, remaining);
   329→
   330→    setIsUploading(true);
   331→    try {
   332→      const formData = new FormData();
   333→      filesToUpload.forEach((file) => formData.append('files', file));
   334→
   335→      const res = await fetch('/api/admin/upload', {
   336→        method: 'POST',
   337→        body: formData,
   338→      });
   339→
   340→      const data = await res.json();
   341→
   342→      if (res.ok && data.urls && data.urls.length > 0) {
   343→        setForm((prev) => ({
   344→          ...prev,
   345→          images: [...prev.images, ...data.urls],
   346→        }));
   347→        toast.success(`Uploaded ${data.urls.length} image(s)`);
   348→      }
   349→
   350→      if (data.errors && data.errors.length > 0) {
   351→        data.errors.forEach((err: string) => toast.error(err));
   352→      }
   353→    } catch {
   354→      toast.error('Upload failed. Please try again.');
   355→    } finally {
   356→      setIsUploading(false);
   357→      if (fileInputRef.current) fileInputRef.current.value = '';
   358→    }
   359→  };
   360→
   361→  const removeImage = (index: number) => {
   362→    setForm((prev) => ({ ...prev, images: prev.images.filter((_, i) => i !== index) }));
   363→  };
   364→
   365→  // ---- Variant management ----
   366→  const addVariant = () => {
   367→    setForm((prev) => ({
   368→      ...prev,
   369→      variants: [...prev.variants, { name: '', color: '', size: '', material: '', price: '', stock: '10' }],
   370→    }));
   371→  };
   372→  const removeVariant = (index: number) => {
   373→    setForm((prev) => ({ ...prev, variants: prev.variants.filter((_, i) => i !== index) }));
   374→  };
   375→  const updateVariant = (index: number, field: keyof VariantRow, value: string) => {
   376→    setForm((prev) => ({
   377→      ...prev,
   378→      variants: prev.variants.map((v, i) => (i === index ? { ...v, [field]: value } : v)),
   379→    }));
   380→  };
   381→
   382→  // ---- Attribute management ----
   383→  const addAttribute = () => {
   384→    setForm((prev) => ({ ...prev, attributes: [...prev.attributes, { name: '', value: '' }] }));
   385→  };
   386→  const removeAttribute = (index: number) => {
   387→    setForm((prev) => ({ ...prev, attributes: prev.attributes.filter((_, i) => i !== index) }));
   388→  };
   389→  const updateAttribute = (index: number, field: keyof AttributeRow, value: string) => {
   390→    setForm((prev) => ({
   391→      ...prev,
   392→      attributes: prev.attributes.map((a, i) => (i === index ? { ...a, [field]: value } : a)),
   393→    }));
   394→  };
   395→
   396→  // ---- Tag toggle ----
   397→  const toggleTag = (tagId: string) => {
   398→    setForm((prev) => ({
   399→      ...prev,
   400→      selectedTags: prev.selectedTags.includes(tagId)
   401→        ? prev.selectedTags.filter((t) => t !== tagId)
   402→        : [...prev.selectedTags, tagId],
   403→    }));
   404→  };
   405→
   406→  const selectedCategory = categories.find((c) => c.id === form.categoryId);
   407→  const subCategories = selectedCategory?.subCategories || [];
   408→
   409→  // ---- Brand form ----
   410→  const [brandForm, setBrandForm] = useState({ name: '', description: '', country: '', foundedYear: '' });
   411→  const [showBrandForm, setShowBrandForm] = useState(false);
   412→
   413→  const handleCreateBrand = async () => {
   414→    if (!brandForm.name.trim()) { toast.error('Brand name is required'); return; }
   415→    try {
   416→      const res = await fetch('/api/admin/brands', {
   417→        method: 'POST',
   418→        headers: { 'Content-Type': 'application/json' },
   419→        body: JSON.stringify(brandForm),
   420→      });
   421→      if (res.ok) {
   422→        toast.success('Brand created successfully');
   423→        const newBrands = await (await fetch('/api/brands')).json();
   424→        setBrands(newBrands);
   425→        setBrandForm({ name: '', description: '', country: '', foundedYear: '' });
   426→        setShowBrandForm(false);
   427→      }
   428→    } catch { toast.error('Error creating brand'); }
   429→  };
   430→
   431→  // ---- Category form ----
   432→  const [categoryForm, setCategoryForm] = useState({ name: '', description: '' });
   433→  const [showCategoryForm, setShowCategoryForm] = useState(false);
   434→
   435→  const handleCreateCategory = async () => {
   436→    if (!categoryForm.name.trim()) { toast.error('Category name is required'); return; }
   437→    try {
   438→      const res = await fetch('/api/admin/categories', {
   439→        method: 'POST',
   440→        headers: { 'Content-Type': 'application/json' },
   441→        body: JSON.stringify(categoryForm),
   442→      });
   443→      if (res.ok) {
   444→        toast.success('Category created successfully');
   445→        const newCategories = await (await fetch('/api/categories')).json();
   446→        setCategories(newCategories);
   447→        setCategoryForm({ name: '', description: '' });
   448→        setShowCategoryForm(false);
   449→      }
   450→    } catch { toast.error('Error creating category'); }
   451→  };
   452→
   453→  // ============================================================
   454→  // RENDER
   455→  // ============================================================
   456→  return (
   457→    <div className="min-h-screen bg-background">
   458→      {/* Header */}
   459→      <div className="bg-cream border-b">
   460→        <div className="max-w-7xl mx-auto px-4 py-8">
   461→          <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4">
   462→            <div className="flex items-center gap-4">
   463→              <button onClick={() => navigateTo('home')} className="p-2 hover:text-gold transition-colors">
   464→                <ArrowLeft className="h-5 w-5" />
   465→              </button>
   466→              <div>
   467→                <h1 className="text-2xl font-light tracking-wide flex items-center gap-3">
   468→                  <Settings className="h-5 w-5 text-gold" />
   469→                  Admin Panel
   470→                </h1>
   471→                <p className="text-xs text-warm-gray mt-1">Manage products, brands, and categories</p>
   472→              </div>
   473→            </div>
   474→            <div className="flex items-center gap-2">
   475→              <Button
   476→                onClick={() => { resetForm(); setShowForm(true); setTimeout(() => formRef.current?.scrollIntoView({ behavior: 'smooth' }), 100); }}
   477→                className="tracking-[0.15em] uppercase text-xs bg-gold text-white hover:bg-gold-dark"
   478→              >
   479→                <Plus className="mr-2 h-4 w-4" />
   480→                Add Product
   481→              </Button>
   482→              <Button
   483→                variant="outline"
   484→                size="sm"
   485→                onClick={logoutAdmin}
   486→                className="tracking-wider uppercase text-xs text-warm-gray hover:text-destructive"
   487→              >
   488→                <LogOut className="mr-1.5 h-3.5 w-3.5" />
   489→                Logout
   490→              </Button>
   491→            </div>
   492→          </div>
   493→        </div>
   494→      </div>
   495→
   496→      <div className="max-w-7xl mx-auto px-4 py-6">
   497→        {/* Tabs */}
   498→        <div className="flex gap-1 border-b mb-6">
   499→          {([
   500→            { key: 'products' as const, label: 'Products', icon: Package, count: adminTotal },
   501→            { key: 'brands' as const, label: 'Brands', icon: Sparkles, count: brands.length },
   502→            { key: 'categories' as const, label: 'Categories', icon: Layers, count: categories.length },
   503→          ]).map((tab) => (
   504→            <button
   505→              key={tab.key}
   506→              onClick={() => setActiveTab(tab.key)}
   507→              className={`flex items-center gap-2 px-4 py-3 text-xs tracking-[0.15em] uppercase border-b-2 transition-colors ${
   508→                activeTab === tab.key
   509→                  ? 'border-gold text-foreground font-medium'
   510→                  : 'border-transparent text-warm-gray hover:text-foreground'
   511→              }`}
   512→            >
   513→              <tab.icon className="h-3.5 w-3.5" />
   514→              {tab.label}
   515→              <Badge variant="secondary" className="ml-1 text-[10px] h-4 px-1.5">
   516→                {tab.count}
   517→              </Badge>
   518→            </button>
   519→          ))}
   520→        </div>
   521→
   522→        {/* ===================== PRODUCTS TAB ===================== */}
   523→        {activeTab === 'products' && (
   524→          <div className="space-y-4">
   525→            {/* Product Form */}
   526→            <AnimatePresence>
   527→              {showForm && (
   528→                <motion.div
   529→                  ref={formRef}
   530→                  initial={{ opacity: 0, height: 0 }}
   531→                  animate={{ opacity: 1, height: 'auto' }}
   532→                  exit={{ opacity: 0, height: 0 }}
   533→                  className="border rounded-sm overflow-hidden"
   534→                >
   535→                  <div className="bg-cream px-6 py-4 flex items-center justify-between border-b">
   536→                    <h3 className="text-sm tracking-wider uppercase font-medium">
   537→                      {editingId ? 'Edit Product' : 'New Product'}
   538→                    </h3>
   539→                    <button onClick={resetForm} className="p-1 hover:text-destructive">
   540→                      <X className="h-4 w-4" />
   541→                    </button>
   542→                  </div>
   543→
   544→                  <div className="p-6 space-y-6">
   545→                    {/* Basic Info */}
   546→                    <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   547→                      <div className="md:col-span-2">
   548→                        <label className="block text-xs tracking-wider uppercase text-warm-gray mb-1.5">Product Name *</label>
   549→                        <Input
   550→                          value={form.name}
   551→                          onChange={(e) => setForm((p) => ({ ...p, name: e.target.value }))}
   552→                          placeholder="e.g. Gucci Horsebit 1955 Loafer"
   553→                          className="h-10"
   554→                        />
   555→                      </div>
   556→                      <div className="md:col-span-2">
   557→                        <label className="block text-xs tracking-wider uppercase text-warm-gray mb-1.5">Description</label>
   558→                        <Textarea
   559→                          value={form.description}
   560→                          onChange={(e) => setForm((p) => ({ ...p, description: e.target.value }))}
   561→                          placeholder="Detailed product description..."
   562→                          rows={3}
   563→                        />
   564→                      </div>
   565→                      <div>
   566→                        <label className="block text-xs tracking-wider uppercase text-warm-gray mb-1.5">Price (₹) *</label>
   567→                        <Input
